在 Vite 中使用无界微前端实现共享依赖，配置方式与 Webpack 有所不同。Vite 的构建机制基于 Rollup，其配置更加简洁。下面我将详细介绍完整的 Vite 配置方案。

## 一、Vite 子应用配置共享依赖

### 1. 子应用 Vite 配置

### 2. 更完善的 Vite 配置方案

```javascript

import { defineConfig, loadEnv } from 'vite'
import vue from '@vitejs/plugin-vue'
import { visualizer } from 'rollup-plugin-visualizer'

export default defineConfig(({ mode }) => {
  const env = loadEnv(mode, process.cwd(), '')
  const isMicro = env.VITE_MICRO_MODE === 'true' // 通过环境变量区分运行模式

  return {
    plugins: [
      vue(),
      // 打包分析（可选）
      mode === 'analyze' &&
        visualizer({
          open: true,
          filename: 'dist/stats.html'

      port: 3001,
      headers: {
        'Access-Control-Allow-Origin': '*',
        'Access-Control-Allow-Methods': '*',
        'Access-Control-Allow-Headers': '*'
      },
      cors: true
    },

    // 构建配置
    build: {
      // 微前端模式下使用 external
      ...(isMicro && {
        rollupOptions: {
          external: [
            'vue',
            'vue-router',
            'vuex',
            'pinia',
            'ant-design-vue',
            'lodash',
            'moment',
            'axios',
            'shared-components' // 自定义共享组件

          output: {
            globals: {
              vue: 'Vue',
              'vue-router': 'VueRouter',
              vuex: 'Vuex',
              pinia: 'Pinia',
              'ant-design-vue': 'antd',
              lodash: '_',
              moment: 'moment',
              axios: 'axios',
              'shared-components': 'SharedComponents'
            }
          }
        }
      }),

      // 公共配置
      sourcemap: true,
      minify: 'terser',
      terserOptions: {
        compress: {
          drop_console: mode === 'production',
          drop_debugger: true
        }
      }
    },

    // 解析配置
    resolve: {
      alias: {
        '@': '/src',
        shared: isMicro ? '/src/externals' : '/src/shared'

    }
```

### 4. 智能依赖加载器（适配 Vite）

```javascript
// sub-app/src/utils/vite-dependency-loader.js
/**
 * Vite 环境下的智能依赖加载器
 * 支持：1. 微前端环境从主应用获取依赖
 *      2. 独立运行环境动态导入依赖
 *      3. 开发环境特殊处理
 */
class ViteDependencyLoader {
  constructor() {
    this.isMicro = !!window.__POWERED_BY_WUJIE__ || import.meta.env.VITE_MICRO_MODE === 'true'
    this.sharedFromMain = window.__SHARED_DEPS_FROM_MAIN__
  }

  /**
   * 加载第三方库
   */
  async loadLibrary(libName) {
    // 1. 微前端模式：从主应用获取
    if (this.isMicro && this.sharedFromMain?.libs?.[libName]) {
      console.log(`[Micro Mode] Loading ${libName} from main app`)
      return this.sharedFromMain.libs[libName]
    }

    // 2. 独立运行模式：动态导入
    console.log(`[Standalone Mode] Loading ${libName} dynamically`)

    const libMap = {
      vue: () => import('vue'),
      'vue-router': () => import('vue-router'),
      vuex: () => import('vuex'),
      pinia: () => import('pinia'),
      'ant-design-vue': () => import('ant-design-vue'),
      lodash: () => import('lodash-es'), // 注意：Vite 中通常使用 lodash-es
      moment: () => import('moment'),
      axios: () => import('axios')
    }

    if (libMap[libName]) {
      try {
        const module = await libMap
        return module.default || module
      } catch (error) {
        console.error(`Failed to load ${libName}:`, error)
        throw error
      }
    }

    // 3. 尝试从 window 全局对象获取（CDN 引入的情况）
    const globalLibs = {
      vue: window.Vue,
      'vue-router': window.VueRouter,
      vuex: window.Vuex,
      pinia: window.Pinia,
      'ant-design-vue': window.antd,
      lodash: window._,
      moment: window.moment,
      axios: window.axios
    }

    if (globalLibs[libName]) {
      console.log(`[Global] Loading ${libName} from window`)
      return globalLibs[libName]
    }

    throw new Error(`Library ${libName} not found`)
  }

  /**
   * 加载共享业务组件
   */
  async loadComponent(componentName) {
    // 微前端模式：从主应用获取
    if (this.isMicro && this.sharedFromMain?.components?.[componentName]) {
      return this.sharedFromMain.components[componentName]
    }

    // 独立运行模式
    try {
      // 方案1：从本地共享组件库导入
      const module = await import('@/shared/components')
      return module[componentName]
    } catch (error) {
      // 方案2：从全局共享组件库导入
      if (window.SharedComponents && window.SharedComponents[componentName]) {
        return window.SharedComponents[componentName]
      }
      throw new Error(`Component ${componentName} not found`)
    }
  }

  /**
   * 批量注册 Vue 插件
   */
  async installVuePlugins(app, plugins) {
    for (const plugin of plugins) {
      if (typeof plugin === 'string') {
        const lib = await this.loadLibrary(plugin)
        if (lib && lib.install) {
          app.use(lib)
        }
      } else if (Array.isArray(plugin)) {
        const [libName, options] = plugin
        const lib = await this.loadLibrary(libName)
        if (lib && lib.install) {
          app.use(lib, options)
        }
      } else if (typeof plugin === 'function') {
        plugin(app)
      }
    }
  }
}

export default new ViteDependencyLoader()
```

### 5. 子应用入口文件适配

```javascript
// sub-app/src/main.js
import { createApp } from 'vue'
import App from './App.vue'
import dependencyLoader from './utils/vite-dependency-loader'

async function bootstrap() {
  const app = createApp(App)

  // 1. 加载 Vue 相关依赖
  const VueRouter = await dependencyLoader.loadLibrary('vue-router')
  const Pinia = await dependencyLoader.loadLibrary('pinia')

  // 2. 安装路由
  const router = VueRouter.createRouter({
    history: VueRouter.createWebHistory(import.meta.env.BASE_URL),
    routes: [] // 你的路由配置
  })
  app.use(router)

  // 3. 安装状态管理
  const pinia = Pinia.createPinia()
  app.use(pinia)

  // 4. 安装 UI 库
  await dependencyLoader.installVuePlugins(app, ['ant-design-vue'])

  // 5. 加载并注册共享业务组件
  const sharedComponents = ['UserAvatar', 'CommonDialog', 'DataTable']
  for (const compName of sharedComponents) {
    try {
      const component = await dependencyLoader.loadComponent(compName)
      if (component) {
        app.component(compName, component)
      }
    } catch (error) {
      console.warn(`Failed to load shared component ${compName}:`, error)
    }
  }

  // 6. 挂载应用
  app.mount('#app')

  // 7. 暴露给无界的生命周期
  if (window.__POWERED_BY_WUJIE__) {
    let instance
    window.__WUJIE_MOUNT = () => {
      instance = app
    }
    window.__WUJIE_UNMOUNT = () => {
      instance?.unmount()
    }
  }
}

// 启动应用
bootstrap().catch((error) => {
  console.error('Failed to bootstrap app:', error)
})
```

### 2. 主应用共享依赖管理

```javascript
// main-app/src/shared-deps/index.js
import * as antd from 'ant-design-vue'
import _ from 'lodash'
import moment from 'moment'
import axios from 'axios'

// 导入自定义共享组件
import * as SharedComponents from 'shared-components'

// 创建共享依赖管理器
class SharedDependencyManager {
  constructor() {
    this.deps = {
      libs: {},
      components: {}
    }
  }

  init() {
    // 第三方库
    this.deps.libs = {
      vue: Vue,
      'vue-router': VueRouter,
      vuex: Vuex,
      pinia: Pinia,
      'ant-design-vue': antd,
      lodash: _,
      moment: moment,
      axios: axios
    }

    // 自定义共享组件
    this.deps.components = SharedComponents

    // 挂载到全局
    window.__SHARED_DEPS = this.deps

    console.log('Shared dependencies initialized')
  }

  // 获取依赖
  getDep(name, type = 'libs') {
    return this.deps[type]?.[name]
  }

  // 添加依赖
  addDep(name, dep, type = 'libs') {
    this.deps[type][name] = dep
  }
}

export default new SharedDependencyManager()

// 在主应用入口中初始化
// main-app/src/main.js
import { createApp } from 'vue'
import App from './App.vue'
import sharedDeps from './shared-deps'

const app = createApp(App)

// 初始化共享依赖
sharedDeps.init()

// 注册共享组件到主应用（如果需要）
Object.entries(sharedDeps.deps.components).forEach(([name, component]) => {
  app.component(name, component)
})

app.mount('#app')
```

### 3. 主应用无界插件配置

```javascript
// main-app/src/plugins/wujie-plugin.js
import { bus } from 'wujie'

export const createSharedDepsPlugin = () => {
  return {
    // 在子应用 iframe 加载前注入共享依赖
    jsBeforeLoaders: [
      {
        callback: (window) => {
          // 将主应用的共享依赖注入到子应用
          window.__SHARED_DEPS_FROM_MAIN = window.parent.__SHARED_DEPS

          // 注入一些工具函数
          window.$shared = {
            // 获取共享依赖
            getLib: (name) => window.__SHARED_DEPS_FROM_MAIN?.libs?.[name],
            getComponent: (name) => window.__SHARED_DEPS_FROM_MAIN?.components?.[name],
            // 调用主应用方法
            callMain: (method, ...args) => {
              return bus.$emit('call-main', { method, args })
            }
          }
        }
      }
    ]
  }
}

// 在主应用中使用
import { createSharedDepsPlugin } from './plugins/wujie-plugin'

// 在加载子应用时
<WujieVue
  name="sub-app"
  url="http://localhost:3001"
  :plugins="[createSharedDepsPlugin()]"
  :props="{
    getSharedLib: (name) => window.__SHARED_DEPS?.libs?.[name],
    getSharedComponent: (name) => window.__SHARED_DEPS?.components?.[name]
  }"
/>
```

## 七、关键注意事项

1. **版本一致性**：确保主应用和子应用使用的共享依赖版本完全一致
2. **Tree-shaking**：Vite 默认支持 tree-shaking，但 external 配置会影响 tree-shaking
3. **热更新**：在微前端模式下，共享组件库的热更新需要主应用重启
4. **性能监控**：监控共享依赖的加载时间和内存使用
5. **错误处理**：增加完善的错误处理和降级方案
6. **按需加载**：对于大型组件库，考虑按需加载配置

通过以上配置，您可以在 Vite 项目中实现与无界微前端的完美集成，同时享受 Vite 的开发速度和构建性能优势。
